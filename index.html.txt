function calcDiscount(subtotal, totalItems) {
  if (totalItems < discSettings.minItems) return 0;
  const rawPct = totalItems * discSettings.percent;

  if (totalItems < discSettings.minItems) return 0;

}

    photo:document.getElementById('adminPhoto').value.trim(),
    emoji:document.getElementById('adminEmoji').value.trim()||'🍽',
  };
  if(id){
    const idx=foods.findIndex(f=>f.id===parseInt(id));
    if(idx>-1)foods[idx]=item;
    showNotif('✅ Item updated — customers will see change instantly!');
  } else {
    foods.push(item);
    showNotif('✅ Item added — customers will see it instantly!');
  }
  saveFoodsData();
  clearAdminForm(); renderAdminList(); renderCategories(); renderFoods();
}

// ══ QR CODE ═══════════════════════════════════════════
let activeQrMode = 'customer';
function showQR(mode) {
  activeQrMode = mode;
  document.querySelectorAll('.qr-tab').forEach(t=>t.classList.remove('active'));
  event.target.classList.add('active');
  generateQR();
}

function generateQR() {
  const el=document.getElementById('qrcode');
  el.innerHTML='';
  const url = activeQrMode==='admin' ? ADMIN_URL : CUSTOMER_URL;
  document.getElementById('qrUrlBox').textContent = url;
  try {
    new QRCode(el,{text:url,width:220,height:220,colorDark:'#E84545',colorLight:'#ffffff',correctLevel:QRCode.CorrectLevel.H});
  } catch(e) {
    el.innerHTML=`<p style="font-size:.85rem;padding:1rem;color:var(--muted);">${url}</p>`;
  }
}

// ══ VIEW SWITCH ═══════════════════════════════════════
function switchView(name) {
  document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(b=>b.classList.remove('active'));
  document.getElementById('view-'+name).classList.add('active');
  const map={customer:'btnCustomer',admin:'btnAdmin',qr:'btnQr'};
  const btn=document.getElementById(map[name]);
  if(btn)btn.classList.add('active');
  if(name==='admin')renderAdminList();
  if(name==='qr')generateQR();
}

// ══ LOGO UPLOAD ══════════════════════════════════════
function handleLogoUpload(e) {
  const file = e.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(ev) {
    const dataUrl = ev.target.result;
    localStorage.setItem('tamro_logo', dataUrl);
    applyLogo(dataUrl);
    showNotif('✅ Logo updated!');
  };
  reader.readAsDataURL(file);
}

function applyLogo(dataUrl) {
  const img = document.getElementById('logoImg');
  const txt = document.getElementById('logoText');
  const preview = document.getElementById('logoPreviewAdmin');
  if (dataUrl) {
    img.src = dataUrl;
    img.style.display = 'block';
    txt.style.display = 'none';
    if (preview) {
      preview.innerHTML = '<img src="'+dataUrl+'" style="width:100%;height:100%;object-fit:contain;">';
    }
  } else {
    img.style.display = 'none';
    txt.style.display = '';
    if (preview) {
      preview.innerHTML = '<span style="font-size:1.8rem;color:var(--muted);">📷</span>';
    }
  }
}

function removeLogo() {
  localStorage.removeItem('tamro_logo');
  applyLogo(null);
  document.getElementById('logoUpload').value = '';
  showNotif('Logo removed');
}
